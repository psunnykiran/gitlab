SET 5 – HOSPITAL MANAGEMENT

New Item → Pipeline → Pipeline Script

pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                git branch: 'main',
                    url: 'YOUR_GITHUB_URL'

                bat 'mvn clean compile'
            }
        }

        stage('Test') {
            steps {
                bat 'mvn test'
            }
        }
    }
}

Save → Build Now → Verify stages and Console Output.


SET 6 – CAMPUS EVENT MANAGEMENT

target/*.jar

CHECK

kubectl version --client

START MINIKUBE

minikube start

CHECK NODES

kubectl get nodes

CREATE NGINX DEPLOYMENT

kubectl create deployment mynginx --image=nginx

CHECK

kubectl get deployments

kubectl get pods

SCALE TO 3

kubectl scale deployment mynginx --replicas=3

VERIFY

kubectl get deployments

kubectl get pods

SCALE TO 5

kubectl scale deployment mynginx --replicas=5

VERIFY

kubectl get pods

DELETE

kubectl delete deployment mynginx

# EC2 UBUNTU TERMINAL

CHECK UBUNTU VERSION

cat /etc/os-release


JAVA INSTALLATION

sudo apt update

sudo apt install openjdk-17-jdk -y

java -version


TOMCAT 9 – UBUNTU 22.04

sudo apt install tomcat9 -y

sudo systemctl enable tomcat9

sudo systemctl start tomcat9

sudo systemctl status tomcat9


TOMCAT 10 – UBUNTU 24.04

sudo apt install tomcat10 -y

sudo systemctl enable tomcat10

sudo systemctl start tomcat10

sudo systemctl status tomcat10


VERIFY TOMCAT

curl -I http://localhost:8080


# WINDOWS CMD – LOCAL LAPTOP

mvn clean package

dir target\*.war

UPLOAD WAR

scp -i "key.pem" target\app.war ubuntu@PUBLIC_IP:/home/ubuntu/


# EC2 UBUNTU TERMINAL

TOMCAT 9 DEPLOYMENT

sudo cp /home/ubuntu/app.war /var/lib/tomcat9/webapps/ROOT.war

sudo systemctl restart tomcat9


TOMCAT 10 DEPLOYMENT

sudo cp /home/ubuntu/app.war /var/lib/tomcat10/webapps/ROOT.war

sudo systemctl restart tomcat10


CHECK

sudo systemctl status tomcat9

OR

sudo systemctl status tomcat10


# BROWSER

http://PUBLIC_IP:8080/


========================================
IMPORTANT REPLACEMENTS
========================================

YOUR_GITHUB_URL = Actual GitHub repository URL

YOUR_USERNAME = Docker Hub username

22cs001 = Example roll number; replace with actual roll number

yourapp.jar = Actual JAR filename from target

yourapp.war = Actual WAR filename from target

PUBLIC_IP = EC2 Public IPv4 address

key.pem = Downloaded AWS private key

main = Replace with master if required

8080 = Application's actual listening port
